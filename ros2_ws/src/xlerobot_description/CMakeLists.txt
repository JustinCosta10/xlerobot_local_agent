cmake_minimum_required(VERSION 3.8)
project(xlerobot_description)
find_package(ament_cmake REQUIRED)
set(XLEROBOT_DD_MESH_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../xlerobot_isaac_sim/source/meshes" CACHE PATH "xlerobot_dd mesh directory")
include(meshes.cmake)
foreach(mesh IN LISTS XLEROBOT_MESHES)
  install(FILES "${XLEROBOT_DD_MESH_DIR}/${mesh}" DESTINATION share/${PROJECT_NAME}/meshes)
endforeach()
install(DIRECTORY urdf config DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/generate_description.py DESTINATION share/${PROJECT_NAME}/scripts)
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(description test/test_description.py)
endif()
ament_package()
