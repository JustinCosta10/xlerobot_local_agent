cmake_minimum_required(VERSION 3.8)
project(xlerobot_control)
find_package(ament_cmake REQUIRED)
find_package(controller_interface REQUIRED)
find_package(diff_drive_controller 4.42.1 REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
add_library(xlerobot_diff_drive_controller SHARED src/diff_drive_controller.cpp)
target_compile_features(xlerobot_diff_drive_controller PUBLIC cxx_std_17)
target_compile_options(xlerobot_diff_drive_controller PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(xlerobot_diff_drive_controller PRIVATE include)
ament_target_dependencies(xlerobot_diff_drive_controller
  controller_interface diff_drive_controller pluginlib rclcpp rclcpp_lifecycle)
pluginlib_export_plugin_description_file(controller_interface controller_plugins.xml)
install(TARGETS xlerobot_diff_drive_controller DESTINATION lib)
install(PROGRAMS scripts/cmd_vel_adapter.py DESTINATION lib/${PROJECT_NAME} RENAME cmd_vel_adapter)
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(diff_drive_timeout test/test_diff_drive_timeout.py TIMEOUT 90)
endif()
ament_package()
